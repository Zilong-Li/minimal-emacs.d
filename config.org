This document is the complete personal layer on top of [[https://github.com/jamescherti/minimal-emacs.d][minimal-emacs.d]].
Tangling it writes every personal file of this directory:

#+begin_src sh :tangle no
emacs --batch -l org config.org -f org-babel-tangle
#+end_src

or =C-c C-v t= in this buffer.  The upstream =early-init.el=, =init.el= and
=README.md= are not part of it: they stay untouched so upstream updates can be
applied independently.  minimal-emacs.d loads the files below at these points:

| File                 | Loaded                                   |
|----------------------+------------------------------------------|
| =pre-early-init.el=  | at the start of =early-init.el=          |
| =post-early-init.el= | at the end of =early-init.el=            |
| =pre-init.el=        | at the start of =init.el=                |
| =post-init.el=       | at the end of =init.el=; loads =lisp/=   |

* Design

- Emacs 31 or newer.  Built-in features come first; third-party packages are
  limited to the key UI, editing and workflow tools: meow, meow-tree-sitter,
  nano-theme, nano-modeline, vertico, orderless, marginalia, consult, corfu,
  tempel, avy, persp-mode, magit, ess, pdf-tools, gptel, codex-ide and ghostel.
- Packages are installed by the built-in package.el (=:ensure t=) or
  package-vc (=:vc=).  Startup activates them from the quickstart file without
  loading package.el at all.
- Everything that is not needed to draw the first frame is deferred with
  =:bind=, =:hook=, =:commands= or =with-eval-after-load=.
- Settings that minimal-emacs.d's =init.el= already provides are not repeated.

* Early startup choices
:PROPERTIES:
:header-args+: :tangle pre-early-init.el
:END:

Read before the first frame exists, so frame parameters set here avoid any
resize or flicker.  The macOS part lets libgccjit find Homebrew GCC's
=libemutls_w.a= for native compilation; it follows Homebrew's =opt/gcc=
symlink, so it survives GCC upgrades.

#+begin_src emacs-lisp
;;; pre-early-init.el --- Personal startup choices -*- no-byte-compile: t; lexical-binding: t; -*-

(setq minimal-emacs-package-initialize-and-refresh nil ; See pre-init.el
      minimal-emacs-disable-mode-line-during-startup nil
      minimal-emacs-frame-title-format "OneEmacs"
      vc-handled-backends '(Git))

;; Set these before creating the first graphical frame (also covers daemons).
(dolist (parameter '((width . 81) (height . 45)
                     (internal-border-width . 12)
                     (left-fringe . 0) (right-fringe . 0)
                     (alpha-background . 0.7)
                     (ns-background-blur . 30)
                     (ns-alpha-elements . (ns-alpha-all))))
  (add-to-list 'default-frame-alist parameter))

(when (eq system-type 'darwin)
  (setq ns-use-proxy-icon nil)

  ;; Homebrew GCC keeps the emulated TLS library below a target/version
  ;; directory.  libgccjit needs that directory when linking native code,
  ;; including the trampolines used to advise primitives such as recursive-edit.
  (when-let* ((library
               (car (append
                     (file-expand-wildcards
                      "/opt/homebrew/opt/gcc/lib/gcc/current/gcc/*/*/libemutls_w.a")
                     (file-expand-wildcards
                      "/usr/local/opt/gcc/lib/gcc/current/gcc/*/*/libemutls_w.a"))))
              (directory (directory-file-name (file-name-directory library)))
              (paths (split-string (or (getenv "LIBRARY_PATH") "")
                                   path-separator t))
              ((not (member directory paths))))
    (setenv "LIBRARY_PATH" (mapconcat #'identity (cons directory paths) path-separator))))

;;; pre-early-init.el ends here
#+end_src

* Package policy
:PROPERTIES:
:header-args+: :tangle post-early-init.el
:END:

Package archives and package.el behaviour.  MELPA comes first so packages
track their development snapshots.  codex-ide is installed but excluded from
activation: its autoloads define Transient menus, which would load Transient
on every start (it is loaded on first use instead, see [[*LLM][LLM]]).

#+begin_src emacs-lisp
;;; post-early-init.el --- Package policy -*- no-byte-compile: t; lexical-binding: t; -*-

;; Built-in package.el installs the few third-party packages.  They opt in
;; explicitly with `:ensure' or `:vc', so built-in libraries are never fetched.
(setq use-package-always-ensure nil
      package-quickstart t
      package-install-upgrade-built-in nil
      ;; Prefer MELPA's development snapshots (meow, gptel, ghostel, ...).
      package-archive-priorities '(("melpa" . 99) ("gnu" . 80) ("nongnu" . 70)))

;; codex-ide's autoloads define Transient menus, which would load Transient at
;; every startup.  Leave it inactive; my-tools.el loads it on first use.
(setq package-load-list '((codex-ide nil) all))

;;; post-early-init.el ends here
#+end_src

* Package activation
:PROPERTIES:
:header-args+: :tangle pre-init.el
:END:

Activating packages from =package-quickstart-file= takes milliseconds, while
=package-initialize= parses all archive contents.  minimal-emacs.d's own
initialization is therefore disabled in =pre-early-init.el=.  use-package
normally loads package.el for every =:ensure= and =:vc= check; the cheap
=my/package-installed-p= avoids that, so package.el is only loaded when a
declared package is actually missing.

#+begin_src emacs-lisp
;;; pre-init.el --- Personal load path -*- no-byte-compile: t; lexical-binding: t; -*-

(when (version< emacs-version "31")
  (error "This personal configuration requires Emacs 31 or newer"))

(add-to-list 'load-path (expand-file-name "lisp" user-emacs-directory))

;; Activate installed packages from `package-quickstart-file' instead of the
;; much slower `package-initialize'.  package.el itself is only loaded when a
;; declared package is missing and has to be installed.
(package-activate-all)

(defun my/package-installed-p (name)
  "Return non-nil if package NAME is installed, without loading package.el."
  (or (memq name (bound-and-true-p package-activated-list))
      (file-directory-p (locate-user-emacs-file (format "elpa/%s" name)))))

(setq use-package-ensure-function
      (lambda (name args state &optional no-refresh)
        (unless (my/package-installed-p (if (eq (car args) t) name (car args)))
          (use-package-ensure-elpa name args state no-refresh))))
(advice-add 'use-package-vc-install :before-until
            (lambda (arg &rest _) (my/package-installed-p (car arg))))

;;; pre-init.el ends here
#+end_src

(use-package recentf
  :defer 0.5

** Theme and fonts

NANO light theme with personal fonts.  In terminal frames the theme's
colours are applied explicitly, including the hardware cursor colour via
OSC 12, so a dark terminal background cannot win.

#+begin_src emacs-lisp
;;; Theme and fonts

(use-package nano-theme
  :ensure t
  :demand t
  :init
  (setq nano-fonts-use nil)
  :config
  (load-theme 'nano-light t)

  (defun my/nano-terminal-colors (&optional frame)
    "Apply NANO's explicit light colors to terminal FRAME.
Set the default face directly so the terminal's own dark background
cannot select the dark face palette."
    (let ((frame (or frame (selected-frame))))
      (when (and (not (display-graphic-p frame))
                 (memq 'nano-light custom-enabled-themes))
        (set-frame-parameter frame 'background-mode 'light)
        (set-face-attribute 'default frame
                            :foreground nano-light-foreground
                            :background nano-light-background)
        (frame-set-background-mode frame)
        (set-face-attribute 'cursor frame
                            :background nano-light-foreground
                            :foreground nano-light-background)
        (set-frame-parameter frame 'cursor-color nano-light-foreground)
        ;; TTY hardware cursors are drawn by the terminal, outside Emacs faces.
        ;; OSC 12 sets their color in xterm-compatible terminals.
        (when (and (not noninteractive)
                   (string-match-p
                    "\\`\\(?:xterm\\|rxvt\\|screen\\|tmux\\|foot\\|alacritty\\|kitty\\|wezterm\\|ghostty\\)"
                    (or (tty-type frame) "")))
          (let* ((terminal (frame-terminal frame))
                 (set-string (format "\e]12;%s\a" nano-light-foreground))
                 (previous (terminal-parameter terminal 'my/nano-cursor-set-string)))
            ;; Let Emacs restore the terminal on exit, client detach, and
            ;; suspension, then reapply the color when the TTY is resumed.
            (set-terminal-parameter
             terminal 'tty-mode-reset-strings
             (cons "\e]112\a"
                   (delete "\e]112\a"
                           (terminal-parameter terminal 'tty-mode-reset-strings))))
            (set-terminal-parameter
             terminal 'tty-mode-set-strings
             (cons set-string
                   (delete set-string
                           (delete previous
                                   (terminal-parameter terminal 'tty-mode-set-strings)))))
            (set-terminal-parameter terminal 'my/nano-cursor-set-string set-string)
            (send-string-to-terminal set-string frame))))))

  (defun my/nano-fonts (&optional frame)
    "Apply personal typography to FRAME, including newly created GUI frames."
    (when (display-graphic-p frame)
      (set-face-attribute 'default frame
                          :family "Roboto Mono" :weight 'light :height 160)
      (set-face-attribute 'bold frame :weight 'regular)
      (set-face-attribute 'italic frame
                          :family "Iosevka" :weight 'light :slant 'italic)
      (set-face-attribute 'variable-pitch frame
                          :family "ETBembo" :weight 'regular :height 240)
      (with-selected-frame (or frame (selected-frame))
        (set-fontset-font t 'unicode
                          (font-spec :family "RobotoMono Nerd Font"
                                     :weight 'light :height 160))
        (set-fontset-font t 'emoji
                          (font-spec :family "Apple Color Emoji") nil 'prepend)
        (set-fontset-font t 'emoji
                          (font-spec :family "Noto Color Emoji") nil 'append))))

  (dolist (setup '(my/nano-terminal-colors my/nano-fonts))
    (funcall setup)
    (add-hook 'after-make-frame-functions setup))

  (set-face-attribute 'minibuffer-prompt nil :foreground "pink" :weight 'bold))

;; Nicer glyphs for truncation and wrapping.
(set-display-table-slot standard-display-table 'truncation (make-glyph-code ?…))
(set-display-table-slot standard-display-table 'wrap (make-glyph-code ?↩))
#+end_src

** Header line (NANO) instead of the mode line

The standard mode line is hidden everywhere; nano-modeline draws a header
line for the modes listed here.  =C-c l s= brings the mode line back in the
current buffer.

;;; Header line (NANO) instead of the mode line

(defvar my/mode-line-format (default-value 'mode-line-format)
  "The standard mode line, restored by `my/toggle-mode-line'.")
(setq-default mode-line-format nil)

(defun my/toggle-mode-line ()
  "Toggle the standard Emacs mode line in the current buffer."
  (interactive)
  (setq mode-line-format (unless mode-line-format my/mode-line-format))
  (force-mode-line-update))

(use-package nano-modeline
  :vc (:url "https://github.com/rougier/nano-modeline" :branch "rewrite" :rev :newest)
  :commands nano-modeline
  :init
  (defun my/nano-modeline (fmt)
    "Install the NANO header line described by the variable FMT."
    (require 'nano-modeline)
    (nano-modeline (symbol-value fmt)))

  (pcase-dolist (`(,fmt . ,hooks)
                 '((nano-modeline-format-default prog-mode-hook text-mode-hook)
                   (nano-modeline-format-org-capture org-capture-mode-hook)
                   (nano-modeline-format-gptel gptel-mode-hook)

  (nano-modeline-padding '(0.20 . 0.25))
  :config
  (defun my/nano-modeline-element-pdf-label ()

  (defvar my/nano-modeline-format-pdf-view
    (cons '(my/nano-modeline-element-pdf-label
            nano-modeline-element-space
            nano-modeline-element-buffer-name
            nano-modeline-element-space
            nano-modeline-element-buffer-mode)
          '(my/nano-modeline-element-pdf-page
            nano-modeline-element-window-status
            nano-modeline-element-space))))

** Windows and frames

=display-buffer-alist= places shells and REPLs in bottom side windows
(toggle them with =C-`= or =C-c o o=) and diagnostics at the bottom.
=which-key= lists a prefix's bindings when =C-h= is pressed after it.

#+begin_src emacs-lisp
;;; Windows and frames

(setq window-divider-default-right-width 24
      window-divider-default-places 'right-only
      window-combination-resize nil
      switch-to-buffer-obey-display-actions t
      switch-to-buffer-in-dedicated-window 'pop
      temp-buffer-max-height 8)
(window-divider-mode 1)
(temp-buffer-resize-mode 1)

(defun my/side-window (side slot &rest extra)
  "Return `display-buffer' actions for a side window at SIDE and SLOT."
  `((display-buffer-in-side-window)
    (side . ,side) (slot . ,slot) (window-height . 0.4)
    (preserve-size . (t . t))
    (window-parameters . ((no-other-window . t) (no-delete-other-windows . t)))
    ,@extra))

(setq display-buffer-alist
      `(("\\*Async Shell Command\\*" (display-buffer-no-window))
        ("\\`\\*\\(Warnings\\|Compile-Log\\|Org Links\\)\\*\\'"
         (display-buffer-no-window) (allow-no-window . t))
        ("^\\*eldoc" display-buffer-at-bottom
         (display-buffer-reuse-window display-buffer-pop-up-window))
        ;; Bottom windows (not side windows)
        ((or (derived-mode . flymake-diagnostics-buffer-mode)
             (derived-mode . flymake-project-diagnostics-mode)
             (derived-mode . messages-buffer-mode)
             (derived-mode . backtrace-mode))
         (display-buffer-reuse-mode-window display-buffer-at-bottom)
         (window-height . 0.4) (dedicated . t) (preserve-size . (t . t)))
        ;; Terminals and shells on the right of the bottom side window,
        ;; REPLs on the left.  Toggle them all with C-` or C-c o o.
        ("\\*\\(?:[^*]\\|\\(?:[^*]-\\)+\\)*\\(?:ghostel\\|e?shell\\)\\*"
         ,@(my/side-window 'bottom 1))
        ("\\*\\(?:R\\|[Jj]ulia\\|\\(?:[^*]\\|\\(?:[^*]-\\)+\\)*[Pp]ython\\)\\(?:.*\\)?\\*"
         ,@(my/side-window 'bottom -1))
        ("^\\*.*[Dd]ictionary\\*"
         (display-buffer-in-side-window) (side . bottom) (window-height . 0.35))))

(defun my/toggle-window-dedicated ()
  "Toggle whether the selected window is dedicated to its buffer."
  (interactive)
  (let ((window (selected-window)))
    (set-window-dedicated-p window (not (window-dedicated-p window)))
    (message "Window is %s" (if (window-dedicated-p window) "dedicated" "normal"))))

** Perspectives

One perspective per project (see [[*Projects (one perspective per project)][Projects]]).  Ephemeral buffers are not
saved, and =consult-buffer= gets a source for the current perspective.

#+begin_src emacs-lisp
;;; Perspectives

(use-package persp-mode
  :ensure t
  :bind ("C-c p s" . persp-kill)
  :custom
  (persp-keymap-prefix "s-p")
  (persp-nil-name "Home")
  (persp-autokill-buffer-on-remove 'kill-weak)
  (persp-before-switch-functions
   (list (lambda (&rest _)
           (when (active-minibuffer-window) (keyboard-escape-quit)))))
  :init
  (persp-mode 1)
  :config
  (defun my/persp-skip-save-buffer-p (buffer)
    "Exclude Magit, Ediff and other transient BUFFERs from saved perspectives."
    (with-current-buffer buffer
      (or (string-match-p "\\`\\*?[Ee]diff" (buffer-name))
          (string-prefix-p "magit-" (symbol-name major-mode))
          (derived-mode-p 'magit-mode 'ediff-mode 'comint-mode 'dired-mode))))
  (add-to-list 'persp-filter-save-buffers-functions #'my/persp-skip-save-buffer-p)

  (defun my/persp-buffer-filter-predicate (buffer)
    "Show BUFFER when it belongs to the current perspective.
In the default perspective show only buffers that belong to no other
perspective.  Always hide noisy internal buffers."
    (let ((current (get-current-persp)))
      (unless (string-match-p "^\\*\\(splash\\|Warning\\|Compile-Log\\|Async\\|Backtrace\\)"
                              (buffer-name buffer))
        (if current
            (memq buffer (persp-buffers current))
          (not (cl-some (lambda (p) (and p (memq buffer (persp-buffers p))))
                        (persp-persps)))))))

  (with-eval-after-load 'consult
    (add-to-list 'consult-buffer-sources
                 (list :name     "Persp"
                       :narrow   ?s
                       :category 'buffer
                       :state    #'consult--buffer-state

(defun my/sexp--move-delimiter (closing slurp)
  "Move the CLOSING (or opening) delimiter of the enclosing list.
With SLURP move it outward over the adjacent sexp, otherwise barf the
outermost sexp by moving the delimiter inward."
  (save-excursion
    (if closing (up-list) (backward-up-list))
    (let ((char (if closing (char-before) (char-after))))
      (delete-char (if closing -1 1))
      (if (eq closing slurp) (forward-sexp) (backward-sexp))
      (unless slurp
        (if closing (skip-chars-backward " \t\n") (skip-chars-forward " \t\n")))
      (insert char))))

(defun my/slurp-forward () "Slurp the next sexp." (interactive) (my/sexp--move-delimiter t t))
(defun my/barf-forward () "Barf the last sexp." (interactive) (my/sexp--move-delimiter t nil))
(defun my/slurp-backward () "Slurp the previous sexp." (interactive) (my/sexp--move-delimiter nil t))
(defun my/barf-backward () "Barf the first sexp." (interactive) (my/sexp--move-delimiter nil nil))

(pcase-dolist (`(,name ,open ,close)
               '((round ?\( ?\)) (square ?\[ ?\]) (curly ?{ ?})
                 (angle ?< ?>) (string ?\" ?\") (backquote ?` ?`)))
  (defalias (intern (format "my/wrap-%s" name))
    (lambda () (interactive) (insert-pair 1 open close))
    (format "Wrap the region or the next sexp with %c%c." open close)))

(defun my/unwrap ()
  "Delete the delimiters of the list enclosing point."
  (interactive)
  (save-excursion (backward-up-list) (delete-pair 1)))

(defun my/rewrap (open)
  "Replace the delimiters of the enclosing list with OPEN and its partner."
  (interactive "cRewrap with: ")
  (let ((close (or (cadr (assq open '((?\( ?\)) (?\[ ?\]) (?{ ?}) (?< ?>)))) open)))
    (save-excursion
      (backward-up-list)
      (let ((start (point)))
        (forward-sexp)
        (delete-char -1)
        (insert close)
        (goto-char start)
        (delete-char 1)
        (insert open)))))

  "Move to the end of the contents of the enclosing list."
  (interactive)
  (up-list)
  (backward-char))

(defun my/jump-to-matching-paren ()
  "Jump to the matching paren, like % in Vim."
  (interactive)
  (cond ((looking-at "\\s(") (forward-list))
        ((looking-back "\\s)" 1) (backward-list))
        (t (up-list))))

(defun my/repl-send-line-or-region (repl &optional step)
  "Send the current line or region to REPL; with STEP move to the next line."
  (let* ((proc (or (get-buffer-process repl)
                   (error "No process found for buffer %s" repl)))
         (min (if (use-region-p) (region-beginning) (line-beginning-position)))
         (max (if (use-region-p) (region-end) (line-end-position)))
         (command (buffer-substring min max)))
    (with-current-buffer (process-buffer proc)
      (goto-char (process-mark proc))
      (insert command "\n")
      (set-marker (process-mark proc) (point)))
    (process-send-string proc (concat command "\n"))
    (display-buffer (process-buffer proc))
    (when step
      (goto-char max)
      (forward-line 1))
    (deactivate-mark)))

(defun my/shell-setup ()
  "Plain completion and auto-kill on exit for `shell-mode'."

    (set-process-sentinel proc

                              (kill-buffer (process-buffer process)))))))
